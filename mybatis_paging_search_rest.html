03 / Paging Formula

페이징 로직 공식

offset = (page - 1) × pageSize

totalPages = CEIL(totalCount / pageSize)

hasPrev = page > 1

hasNext = page < totalPages
💡 예시 전체 53건, pageSize=10 일 때 → totalPages=6
page=3 → LIMIT 10 OFFSET 20 → 21~30번째 행 조회

package com.example.board.vo;

import lombok.*;
import jakarta.validation.constraints.*;

@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class SearchVO {

    /** 검색 유형: all | name | title | content */
    @Builder.Default
    private String searchType = "all";

    /** 검색 키워드 */
    private String keyword;

    /** 현재 페이지 (1-based) */
    @Builder.Default
    @Min(1)
    private int page = 1;

    /** 페이지당 건수 */
    @Builder.Default
    @Min(1) @Max(100)
    private int pageSize = 10;

    /** 정렬 컬럼 (화이트리스트 검증 후 ${}로 사용) */
    @Builder.Default
    private String sortColumn = "id";

    @Builder.Default
    private String sortDir = "DESC";

    /** MyBatis LIMIT/OFFSET 계산값 (Service에서 set) */
    public int getOffset() {
        return (page - 1) * pageSize;
    }

    /** 허용된 정렬 컬럼 화이트리스트 */
    public String getSafeSortColumn() {
        return switch (sortColumn) {
            case "title", "writer", "viewCount", "createdAt" -> sortColumn;
            default -> "id";
        };
    }

    public String getSafeSortDir() {
        return "ASC".equalsIgnoreCase(sortDir) ? "ASC" : "DESC";
    }
}
PageResult.java — 페이지 응답 객체Java
package com.example.board.vo;

import lombok.*;
import java.util.List;

@Data
@Builder
public class PageResult<T> {
    private List<T> list;         // 현재 페이지 데이터

    private int     page;          // 현재 페이지
    private int     pageSize;      // 페이지당 건수
    private long    totalCount;    // 전체 건수
    private int     totalPages;    // 전체 페이지 수
    private boolean hasNext;       // 다음 페이지 존재 여부
    private boolean hasPrev;       // 이전 페이지 존재 여부

    private String  searchType;    // 검색 타입 echo
    private String  keyword;       // 검색 키워드 echo

    public static <T> PageResult<T> of(
            List<T> list, long totalCount, SearchVO search) {
        int totalPages = (int) Math.ceil((double) totalCount / search.getPageSize());
        return PageResult.<T>builder()
                .list(list)
                .page(search.getPage())
                .pageSize(search.getPageSize())
                .totalCount(totalCount)
                .totalPages(totalPages)
                .hasNext(search.getPage() < totalPages)
                .hasPrev(search.getPage() > 1)
                .searchType(search.getSearchType())
                .keyword(search.getKeyword())
                .build();
    }
}

package com.example.board.common;

import lombok.*;

@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class ApiResponse<T> {
    private boolean success;
    private String  message;
    private T       data;
    private Integer status;

    public static <T> ApiResponse<T> ok(T data) {
        return ApiResponse.<T>builder()
                .success(true)
                .message("OK")
                .data(data)
                .status(200)
                .build();
    }

    public static <T> ApiResponse<T> created(T data) {
        return ApiResponse.<T>builder()
                .success(true)
                .message("Created")
                .data(data)
                .status(201)
                .build();
    }

    public static <T> ApiResponse<T> error(String message, int status) {
        return ApiResponse.<T>builder()
                .success(false)
                .message(message)
                .status(status)
                .build();
    }

검색 타입(all·name·title·content)과 페이징을 완전 동적 SQL로 처리합니다.

src/main/resources/mapper/BoardMapper.xml XML 
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper
  PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
  "http://mybatis.org/dtd/mybatis-3-mapper.dtd">

<mapper namespace="com.example.board.mapper.BoardMapper">

  <!-- ============= ResultMap ============= -->
  <resultMap id="boardMap" type="BoardVO">
    <id     column="id"         property="id"        />
    <result column="writer"     property="writer"    />
    <result column="title"      property="title"     />
    <result column="content"    property="content"   />
    <result column="view_count" property="viewCount" />
    <result column="created_at" property="createdAt" />
    <result column="updated_at" property="updatedAt" />
  </resultMap>

  <!-- ============= 공통 WHERE 조각 ============= -->
  <sql id="searchCondition">
    <where>
      <if test="keyword != null and keyword != ''">
        <choose>
          <when test="searchType == 'name'">
            <bind name="pat" value="'%' + keyword + '%'"/>
            writer LIKE #{pat}
          </when>
          <when test="searchType == 'title'">
            <bind name="pat" value="'%' + keyword + '%'"/>
            title LIKE #{pat}
          </when>
          <when test="searchType == 'content'">
            <bind name="pat" value="'%' + keyword + '%'"/>
            content LIKE #{pat}
          </when>
          <otherwise>      <!-- all : 이름 OR 제목 OR 내용 -->
            <bind name="pat" value="'%' + keyword + '%'"/>
            (writer LIKE #{pat}
              OR title   LIKE #{pat}
              OR content LIKE #{pat})
          </otherwise>
        </choose>
      </if>
    </where>
  </sql>

  <!-- ============= 목록 조회 (페이징 + 검색) ============= -->
  <select id="findList"
          parameterType="SearchVO"
          resultMap="boardMap">
    SELECT id, writer, title, content, view_count, created_at, updated_at
    FROM   board
    <include refid="searchCondition"/>
    ORDER BY ${safeSortColumn} ${safeSortDir}
    LIMIT  #{pageSize}
    OFFSET #{offset}
  </select>

  <!-- ============= 전체 건수 (같은 WHERE 재사용) ============= -->
  <select id="countTotal"
          parameterType="SearchVO"
          resultType="long">
    SELECT COUNT(*)
    FROM   board
    <include refid="searchCondition"/>
  </select>

  <!-- ============= 단건 조회 ============= -->
  <select id="findById"
          parameterType="long"
          resultMap="boardMap">
    SELECT id, writer, title, content, view_count, created_at, updated_at
    FROM   board
    WHERE  id = #{id}
  </select>

  <!-- ============= 조회수 증가 ============= -->
  <update id="increaseViewCount" parameterType="long">
    UPDATE board SET view_count = view_count + 1 WHERE id = #{id}
  </update>

  <!-- ============= 등록 ============= -->
  <insert id="insert"
          parameterType="BoardVO"
          useGeneratedKeys="true"
          keyProperty="id">
    INSERT INTO board (writer, title, content)
    VALUES (#{writer}, #{title}, #{content})
  </insert>

  <!-- ============= 수정 ============= -->
  <update id="update" parameterType="BoardVO">
    UPDATE board
    <set>
      <if test="title != null and title != ''">   title   = #{title},   </if>
      <if test="content != null and content != ''"> content = #{content}, </if>
    </set>
    WHERE id = #{id}
  </update>

  <!-- ============= 삭제 ============= -->
  <delete id="delete" parameterType="long">
    DELETE FROM board WHERE id = #{id}
  </delete>

</mapper>

package com.example.board.service;

import com.example.board.vo.*;

public interface BoardService {
    PageResult<BoardVO> getList(SearchVO search);
    BoardVO             getDetail(long id);
    BoardVO             write(BoardVO board);
    BoardVO             modify(long id, BoardVO board);
    void                remove(long id);

BoardServiceImpl.javaJava
package com.example.board.service;

import com.example.board.mapper.BoardMapper;
import com.example.board.vo.*;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class BoardServiceImpl implements BoardService {

    private final BoardMapper boardMapper;

    /** 목록 + 페이지 정보 반환 */
    @Override
    public PageResult<BoardVO> getList(SearchVO search) {
        long             total = boardMapper.countTotal(search);
        List<BoardVO>  list  = boardMapper.findList(search);
        return PageResult.of(list, total, search);
    }

    /** 단건 조회 + 조회수 증가 */
    @Override
    @Transactional
    public BoardVO getDetail(long id) {
        boardMapper.increaseViewCount(id);
        return boardMapper.findById(id)
                .orElseThrow(() -> new RuntimeException("게시글을 찾을 수 없습니다. id=" + id));
    }

    /** 등록 */
    @Override
    @Transactional
    public BoardVO write(BoardVO board) {
        boardMapper.insert(board);
        return boardMapper.findById(board.getId()).orElseThrow();
    }

    /** 수정 */
    @Override
    @Transactional
    public BoardVO modify(long id, BoardVO board) {
        board.setId(id);
        int updated = boardMapper.update(board);
        if (updated == 0) throw new RuntimeException("수정 실패. id=" + id);
        return boardMapper.findById(id).orElseThrow();
    }

    /** 삭제 */
    @Override
    @Transactional
    public void remove(long id) {
        int deleted = boardMapper.delete(id);
        if (deleted == 0) throw new RuntimeException("삭제 실패. id=" + id);
    }
}

BoardController.javaJava
package com.example.board.controller;

import com.example.board.common.ApiResponse;
import com.example.board.service.BoardService;
import com.example.board.vo.*;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/boards")
@RequiredArgsConstructor
public class BoardController {

    private final BoardService boardService;

    /**
     * 목록 조회 (페이징 + 검색)
     * GET /api/boards?page=1&pageSize=10&searchType=title&keyword=spring
     */
    @GetMapping
    public ResponseEntity<ApiResponse<PageResult<BoardVO>>> list(
            @ModelAttribute @Valid SearchVO search) {

        PageResult<BoardVO> result = boardService.getList(search);
        return ResponseEntity.ok(ApiResponse.ok(result));
    }

    /**
     * 단건 조회
     * GET /api/boards/{id}
     */
    @GetMapping("/{id}")
    public ResponseEntity<ApiResponse<BoardVO>> detail(@PathVariable long id) {
        return ResponseEntity.ok(ApiResponse.ok(boardService.getDetail(id)));
    }

    /**
     * 등록
     * POST /api/boards
     * Body: { "writer":"홍길동", "title":"제목", "content":"내용" }
     */
    @PostMapping
    public ResponseEntity<ApiResponse<BoardVO>> write(
            @RequestBody @Valid BoardVO board) {
        return ResponseEntity
                .status(HttpStatus.CREATED)
                .body(ApiResponse.created(boardService.write(board)));
    }

    /**
     * 수정
     * PUT /api/boards/{id}
     * Body: { "title":"새제목", "content":"새내용" }
     */
    @PutMapping("/{id}")
    public ResponseEntity<ApiResponse<BoardVO>> modify(
            @PathVariable long id,
            @RequestBody  BoardVO board) {
        return ResponseEntity.ok(ApiResponse.ok(boardService.modify(id, board)));
    }

    /**
     * 삭제
     * DELETE /api/boards/{id}
     */
    @DeleteMapping("/{id}")
    public ResponseEntity<ApiResponse<Void>> remove(@PathVariable long id) {
        boardService.remove(id);
        return ResponseEntity.ok(ApiResponse.<Void>builder()
                .success(true).message("삭제되었습니다.").status(200).build());
    }
}
11 / Exception

GlobalExceptionHandler.javaJava
package com.example.board.common;

import org.springframework.http.*;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.*;

@RestControllerAdvice
public class GlobalExceptionHandler {

    /** Validation 에러 */
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiResponse<?>> handleValidation(
            MethodArgumentNotValidException ex) {
        String msg = ex.getBindingResult()
                       .getFieldErrors()
                       .stream()
                       .map(e -> e.getField() + ": " + e.getDefaultMessage())
                       .findFirst()
                       .orElse("잘못된 요청");
        return ResponseEntity
                .status(HttpStatus.BAD_REQUEST)
                .body(ApiResponse.error(msg, 400));
    }

    /** 게시글 없음 등 비즈니스 에러 */
    @ExceptionHandler(RuntimeException.class)
    public ResponseEntity<ApiResponse<?>> handleRuntime(RuntimeException ex) {
        return ResponseEntity
                .status(HttpStatus.NOT_FOUND)
                .body(ApiResponse.error(ex.getMessage(), 404));
    }
}

MyBatisConfig.java — LocalDateTime TypeHandlerJava
package com.example.board.config;

import org.apache.ibatis.type.*;
import org.mybatis.spring.boot.autoconfigure.ConfigurationCustomizer;
import org.springframework.context.annotation.*;
import java.sql.*;
import java.time.LocalDateTime;

@Configuration
public class MyBatisConfig {

    @Bean
    public ConfigurationCustomizer mybatisConfigurationCustomizer() {
        return configuration -> {
            // LocalDateTime ↔ DATETIME 자동 변환
            configuration.getTypeHandlerRegistry()
                    .register(LocalDateTime.class,
                              JdbcType.TIMESTAMP,
                              new LocalDateTimeTypeHandler());
        };
    }

    /** LocalDateTime TypeHandler */
    public static class LocalDateTimeTypeHandler
            extends BaseTypeHandler<LocalDateTime> {
        @Override
        public void setNonNullParameter(PreparedStatement ps,
                int i, LocalDateTime parameter, JdbcType jdbcType) throws SQLException {
            ps.setTimestamp(i, Timestamp.valueOf(parameter));
        }
        @Override
        public LocalDateTime getNullableResult(ResultSet rs, String columnName)
                throws SQLException {
            Timestamp ts = rs.getTimestamp(columnName);
            return ts == null ? null : ts.toLocalDateTime();
        }
        @Override
        public LocalDateTime getNullableResult(ResultSet rs, int columnIndex)
                throws SQLException {
            Timestamp ts = rs.getTimestamp(columnIndex);
            return ts == null ? null : ts.toLocalDateTime();
        }
        @Override
        public LocalDateTime getNullableResult(CallableStatement cs, int columnIndex)
                throws SQLException {
            Timestamp ts = cs.getTimestamp(columnIndex);
            return ts == null ? null : ts.toLocalDateTime();
        }
    }
}

16 / Best Practices

실전 팁

항목권장 사항
${}` SQL Injection 정렬 컬럼·방향은 반드시 화이트리스트 검증 후 ${} 사용 (getSafeSortColumn())
N+1 방지 목록 API는 content 컬럼 제외 (SUBSTR 또는 별도 VO), 단건 API에서만 전체 조회
LIKE 인덱스 LIKE '%keyword'는 인덱스 미사용 → FULLTEXT INDEX + MATCH AGAINST 고려
COUNT 쿼리 최적화 SELECT COUNT(*)는 동일 WHERE 절로 캐싱 가능; 2차 캐시 또는 Redis로 TTL 적용
페이지 번호 검증 page > totalPages 요청 시 빈 리스트 대신 마지막 페이지로 리다이렉트 고려
Cursor 페이징 데이터가 수백만 건 이상이면 OFFSET 대신 WHERE id < #{lastId} 방식 사용
SQL 로그 log-impl: StdOutImpl은 개발용; 운영에서는 slf4j로 변경
