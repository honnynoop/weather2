기존 Docker 스택 실행 확인

PowerShell / bash

#
# Airflow는 없어도 됩니다 ✅
PostgreSQL + pgvector 확인
docker exec mds_postgres psql -U airflow -d airflow \
  -c "SELECT extversion FROM pg_extension WHERE extname='vector';"
# → 0.7.x 가 출력되어야 함 (없으면 pgvector 설치 필요)
2
weather · analytics 스키마 생성 (최초 1회)
bash
# weather 스키마 (weather_init.sql 이미 실행했다면 스킵)
docker exec mds_postgres psql -U airflow -d airflow \
  -c "CREATE SCHEMA IF NOT EXISTS weather;
      CREATE TABLE IF NOT EXISTS weather.weather_data (
          id SERIAL PRIMARY KEY, city VARCHAR(100), country VARCHAR(10),
          description VARCHAR(200), temperature_c NUMERIC(5,2),
          feels_like_c NUMERIC(5,2), temp_min_c NUMERIC(5,2), temp_max_c NUMERIC(5,2),
          pressure_hpa INTEGER, humidity_pct INTEGER,
          wind_speed_ms NUMERIC(6,2), wind_deg INTEGER,
          visibility_m INTEGER, cloudiness_pct INTEGER,
          time_of_record TIMESTAMP, sunrise_time TIMESTAMP, sunset_time TIMESTAMP,
          collected_at TIMESTAMP DEFAULT NOW()
      );"

# analytics 스키마
docker exec mds_postgres psql -U airflow -d airflow \
  -c "CREATE SCHEMA IF NOT EXISTS analytics;
      CREATE TABLE IF NOT EXISTS analytics.weather_sales_corr (
          id SERIAL PRIMARY KEY, city VARCHAR(100), weather_desc VARCHAR(200),
          avg_temp_c NUMERIC(5,2), total_orders INTEGER DEFAULT 0,
          total_revenue NUMERIC(14,2) DEFAULT 0, top_category VARCHAR(100),
          calc_date DATE DEFAULT CURRENT_DATE, created_at TIMESTAMP DEFAULT NOW(),
          UNIQUE(city, calc_date)
      );
      CREATE TABLE IF NOT EXISTS analytics.embedding_log (
          id SERIAL PRIMARY KEY, source_type VARCHAR(50),
          source_id BIGINT, embedded_at TIMESTAMP DEFAULT NOW(),
          UNIQUE(source_type, source_id)
      );"

Step 5
Superset 대시보드 연동
✓
Superset에서 날씨 데이터 시각화
Superset SQL Lab 확인 쿼리
-- 1. 도시별 기온 트렌드
SELECT DATE_TRUNC('hour', collected_at) AS hour,
       city, AVG(temperature_c) AS avg_temp
FROM weather.weather_data
WHERE collected_at >= NOW() - INTERVAL '7 days'
GROUP BY 1, 2 ORDER BY 1;

-- 2. 날씨별 매출 상관관계 (Batch 실행 후)
SELECT city, weather_desc, avg_temp_c,
       total_orders, total_revenue, top_category
FROM analytics.weather_sales_corr
ORDER BY calc_date DESC, city;

-- 3. RAG 임베딩 인덱스 현황
SELECT COUNT(*) AS embedded_docs FROM public.vector_store;

Troubleshooting
트러블슈팅
!
시작 시 "relation weather.weather_data does not exist"

weather 스키마가 없습니다. Step 1의 스키마 생성 명령을 실행하거나:

bash
docker exec mds_postgres psql -U airflow -d airflow \
  -c "CREATE SCHEMA IF NOT EXISTS weather;"
# 이후 weather_init.sql 실행 또는 Step 1의 CREATE TABLE 명령 실행
!
status API에서 totalRecords = 0

DataSeederRunner가 실패했거나 auto-seed가 꺼진 경우. 수동으로 시딩하세요:

curl
curl -X POST "http://localhost:8090/api/mock/seed?days=7"
!
RAG 채팅 응답이 "관련 데이터를 찾을 수 없습니다"

임베딩 인덱스 미생성. Batch Job 2를 실행하세요:

curl
curl -X POST http://localhost:8090/api/batch/run/embedding-index
# 완료 후 vector_store 테이블 건수 확인
docker exec mds_postgres psql -U airflow -d airflow \
  -c "SELECT COUNT(*) FROM public.vector_store;"
!

spring:
