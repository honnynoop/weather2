🦙 Ollama 모델
  • llama3.2 (3B): 채팅·추천·리포트 생성. ~2GB. CPU에서 약 10~30초/응답
  • nomic-embed-text: 텍스트 → 768차원 벡터 변환. ~274MB. 매우 빠름
  • 저사양 PC: llama3.2:1b (~1.3GB)으로 대체 가능
  • 프로덕션: OpenAI text-embedding-3-small (1536차원)으로 교체 가능
⚙️ application.yml 핵심 설정
  • spring.ai.ollama.chat.model: llama3.2
  • spring.ai.ollama.embedding.model: nomic-embed-text
  • spring.ai.vectorstore.pgvector.dimensions: 768
  • spring.ai.vectorstore.pgvector.initialize-schema: true
  • spring.batch.job.enabled: false → 시작 시 자동실행 방지

// 날씨-매출 JOIN: Seoul↔서울, Busan↔부산 CASE 매핑
String sql = """
    SELECT w.city, w.description AS weather_desc,
           AVG(w.temperature_c) AS avg_temp_c,
           COUNT(DISTINCT o.order_id) AS total_orders,
           COALESCE(SUM(o.quantity * o.unit_price * (1 - o.discount_rate)), 0) AS total_revenue,
           (상위 카테고리 서브쿼리...) AS top_category
    FROM weather.weather_data w
    LEFT JOIN raw.customers c ON c.city LIKE (
        CASE w.city
            WHEN 'Seoul'   THEN '서울%'
            WHEN 'Busan'   THEN '부산%'
            WHEN 'Incheon' THEN '인천%'
            WHEN 'Daegu'   THEN '대구%'
        END)
    LEFT JOIN raw.orders o ON o.customer_id = c.customer_id
        AND o.order_date = CURRENT_DATE AND o.status = 'completed'
    WHERE DATE(w.collected_at) = CURRENT_DATE
    GROUP BY w.city, w.description
    """;

① WeatherRecommendationService — ChatClient 추천

WeatherRecommendationService.java
public String recommend(String city, Long customerId) {
    // 1. 현재 날씨 조회 (weather.weather_data)
    WeatherData weather = weatherRepo
        .findTopByCityOrderByCollectedAtDesc(city).orElseThrow();

    // 2. 고객 구매 이력 (raw.orders + raw.products)
    List<SalesOrder> history = orderRepo
        .findTop10ByCustomer_CustomerIdOrderByOrderDateDesc(customerId);

    // 3. llama3.2에 컨텍스트 전달 → 추천 생성
    return chatClient.prompt()
        .user(u -> u.text("""
            현재 날씨 ({city}): {weather}
            구매 이력: {history}
            날씨에 맞는 상품 3개를 추천하고 이유를 설명해주세요.
            """)
            .param("city", city)
            .param("weather", weather.toText())
            .param("history", historyText))
        .call().content();
}

② WeatherChatService — RAG Q&A

WeatherChatService.java — QuestionAnswerAdvisor
public String chat(String question) {
    return chatClient.prompt()
        // QuestionAnswerAdvisor = RAG 자동 처리
        // ① 질문을 nomic-embed-text로 임베딩
        // ② PgVector에서 코사인 유사도 상위 5개 검색
        // ③ 검색 결과를 컨텍스트로 추가해 llama3.2에 전달
        .advisors(QuestionAnswerAdvisor.builder(vectorStore)
            .searchRequest(SearchRequest.builder()
                .query(question)
                .topK(5)
                .similarityThreshold(0.5)
                .build())
            .build())
        .user(question)
        .call().content();
}
// 도시 필터 버전: filterExpression("city == 'Seoul'") 추가

③ WeatherReportService — AI 리포트 생성

WeatherReportService.java
public String generateReport(int days) {
    // analytics.weather_sales_corr 집계 데이터 조회
    List<WeatherSalesCorr> data = corrRepo.findBetween(
        LocalDate.now().minusDays(days), LocalDate.now());

    String dataText = data.stream()
        .map(WeatherSalesCorr::toText)  // 텍스트 변환
        .collect(Collectors.joining("\n"));

    // llama3.2가 5개 섹션 마크다운 리포트 생성
    return chatClient.prompt()
        .user(u -> u.text("""
            최근 {days}일 날씨-매출 데이터: {data}
            
            1. 핵심 요약  2. 상관관계  3. 도시별 특이사항
            4. 인기 카테고리  5. 마케팅 제안 3가지
            마크다운으로 작성하고 수치를 포함하세요.
            """)
            .param("days", String.valueOf(days))
            .param("data", dataText))
        .call().content();
}
